% === Torsion irreducible decomposition ===
% Hehl et al. Rev.Mod.Phys. 48, 393 (1976) labelling:
%   ^{(1)}T  traceless tensor part   q_{mu nu rho}
%   ^{(2)}T  trace vector part       T_mu = T^alpha_{\ mu alpha}
%   ^{(3)}T  axial pseudovector      S^mu = epsilon^{alpha beta gamma mu} T_{alpha beta gamma}
% Convention: Shapiro hep-th/0103093 §2; BHL 2406.12826; supervisor template
% Usage: \T[1]{_{\mu\nu\rho}}, \T[2]{_\mu}, \T[3]{^\mu}

\newcommand{\T}[2][]{%
  \IfEqCase{#1}{%
    {1}{\accentset{(1)}{T}_{#2}}%
    {2}{\accentset{(2)}{T}_{#2}}%
    {3}{\accentset{(3)}{T}_{#2}}%
    {}{{T}_{#2}}%
  }[\PackageError{macros}{Invalid torsion index: #1}{}]%
}

% Lagrange multiplier components (for constrained PGT action)
\newcommand{\TLambda}[2][]{%
  \IfEqCase{#1}{%
    {1}{\accentset{(1)}{\Lambda}_{#2}}%
    {2}{\accentset{(2)}{\Lambda}_{#2}}%
    {3}{\accentset{(3)}{\Lambda}_{#2}}%
    {}{\Lambda_{#2}}%
  }[\PackageError{macros}{Invalid multiplier index: #1}{}]%
}
