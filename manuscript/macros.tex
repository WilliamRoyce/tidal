% === Supervisor's physics macros ===

% Planck mass
\newcommand{\Planck}{M_{\text{Pl}}}

% Covariant derivatives
\newcommand{\rD}[1]{\accentset{\circ}{\nabla}_{#1}}   % Riemannian
\newcommand{\rcD}[1]{\nabla_{#1}}                       % Riemann-Cartan

% Curvature tensors
\newcommand{\rR}[1]{\accentset{\circ}{R}_{#1}}         % Riemann
\newcommand{\rcR}[1]{R_{#1}}                             % Riemann-Cartan

% Torsion components (use \T[1]{indices}, \T[2]{indices}, \T[3]{indices})
\newcommand{\T}[2][]{%
  \IfEqCase{#1}{%
    {1}{\accentset{(1)}{T}_{#2}}%
    {2}{\accentset{(2)}{T}_{#2}}%
    {3}{\accentset{(3)}{T}_{#2}}%
    {}{{T}_{#2}}%
  }[\PackageError{macros}{Invalid torsion index: #1}{}]%
}

% Multiplier components
\newcommand{\TLambda}[2][]{%
  \IfEqCase{#1}{%
    {1}{\accentset{(1)}{\Lambda}_{#2}}%
    {2}{\accentset{(2)}{\Lambda}_{#2}}%
    {3}{\accentset{(3)}{\Lambda}_{#2}}%
    {}{\Lambda_{#2}}%
  }[\PackageError{macros}{Invalid multiplier index: #1}{}]%
}
