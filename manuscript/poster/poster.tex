% ---- Block template: standard-blue title bar, a THICK warm-blue rule flush at
% its bottom (separating header from body, like the talk frametitle), then a
% padded light-blue body. The Gemini default nests the separator inside the title
% box and pulls it up with \vskip-1.25ex (landing part-way through the bar); we
% move it OUT into its own full-width box so it sits cleanly at the bottom.
\setbeamertemplate{block begin}{
  \begin{beamercolorbox}[colsep*=0ex,dp=1ex,center]{block title}
    \vskip0.4ex
    \usebeamerfont{block title}\insertblocktitle
    \vskip0.4ex
  \end{beamercolorbox}
  \nointerlineskip
  \begin{beamercolorbox}[wd=\linewidth,ht=0.55ex,sep=0pt]{block separator}\end{beamercolorbox}
  \nointerlineskip
  \usebeamerfont{block body}
  \begin{beamercolorbox}[colsep*=0ex]{block body}
    \justifying
    \begin{adjustwidth}{1.5ex}{1.5ex}
      \setlength{\parskip}{1ex} \vskip0.5ex } \setbeamertemplate{block end}{ \vskip0.5ex
    \end{adjustwidth}
  \end{beamercolorbox}
  \vskip0pt
  \vspace*{2ex}
}

% ---- Alert block (cherry key-finding): same separator fix -- thick dark-cherry
% rule flush at the bottom of the cherry title bar (was nested part-way through).

    \end{adjustwidth}
  \end{beamercolorbox}
  \vskip0pt
  \vspace*{2ex}
}

% ---- Header typography: large full-width title ------------------------------
% 124pt keeps line 1 ("Amplifying graviton--photon conversion") on a single line
% within the 0.92\paperwidth title parbox (130 overflowed to a 3rd line); the larger
% \baselineskip (170) opens the gap between the two title lines.
\setbeamerfont{headline title}{size=\fontsize{124}{170}\selectfont,series=\bfseries}
\setbeamerfont{headline author}{size=\LARGE,series=\bfseries}
\setbeamerfont{headline institute}{size=\normalsize}

% ---- Custom headline: two separate strips with a gap --
%   (1) standard-blue strip with the title,
%   (2) gap (page background),
%   (3) light-blue strip with logo + author + affiliation,
%   (4) BOLD warm-blue rule before the body.
\setbeamertemplate{headline}{%
  % (1) standard-blue title strip (full width) -- tall, generous title whitespace
  \begin{beamercolorbox}[wd=\paperwidth,sep=0pt]{headline title strip}%
    \vskip20ex
    \centering
    {\usebeamerfont{headline title}\inserttitle\par}
    \vskip20ex
  \end{beamercolorbox}%
  \nointerlineskip
  % (2) light-blue strip flush below the title strip (no white gap):
  %     logo (left) + author/affiliation (centred)
  \begin{beamercolorbox}[wd=\paperwidth,sep=0pt]{headline}%
    \vskip2.6ex
    \begin{columns}[c]
      \begin{column}{0.20\paperwidth}
        % \mbox forces restricted horizontal mode: a leading \hspace* at the start
        % of a column body otherwise begins a paragraph and TeX discards the leading
        % glue (which is why \hspace{5cm}/{8cm} never moved the logo). Inside an
        % \mbox the kern is always kept, so this reliably insets the logo.
        \mbox{\hspace*{2.5cm}\includegraphics[height=4.0cm]{uc-cmyk}}
      \end{column}
      \begin{column}{0.60\paperwidth}
        \centering
        {\usebeamerfont{headline author}\insertauthor\par}
        \vskip0.9ex
        {\usebeamerfont{headline institute}\insertinstitute\par}
      \end{column}
      \begin{column}{0.20\paperwidth}\mbox{}\end{column}
    \end{columns}
    \vskip2.6ex
  \end{beamercolorbox}%
  \nointerlineskip
  % (4) BOLD warm-blue rule separating header from body
  \begin{beamercolorbox}[wd=\paperwidth,ht=1.1ex,sep=0pt]{headline rule}\end{beamercolorbox}%
}
