% operator landscape that motivates an automated survey.
\begin{block}{1.\quad Gravity with torsion}
  Treating gravity as a gauge theory of the Poincar\'e group adds
  \textbf{torsion} $\MAGT{}$ --- the gauge field of local translations --- as a
  second field strength alongside \textbf{curvature} $\MAGF{}$.
  \vspace{0.6ex}
  \begin{center}
    \includegraphics[width=\linewidth]{pgt_geometry.pdf}
  \end{center}
  \vspace{0.4ex}
  Its most general action holds \emph{many} independent couplings ---
  $\MAGT{^2}$, $\MAGF{^2}$, $\MAGF{}\!\times\!\FieldF{}$, $\MAGF{}\!\times\!\rcD{}\MAGT{}$,
  $\dots$ --- far too many to work through by hand.
\end{block}
