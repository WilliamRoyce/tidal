% --- E.1  Einstein--Maxwell Gertsenshtein baseline ------------------
\subsection{Einstein--Maxwell Gertsenshtein baseline}%
\label{SymbolicBaseline}
The pure Einstein--Maxwell Lagrangian on a uniform magnetic background
$\bm{B} = \Bzero\,\hat{x}$ produces the equations of
\cref{BaselineKineticOutput}. The kinetic sub-matrix is block-diagonal
in the transverse channels.

% --- E.2  Euler--Heisenberg QED 1-loop -------------------------------
\subsection{Euler--Heisenberg QED corrections}%
\label{SymbolicEulerHeisenberg}
The Euler--Heisenberg theory of \cref{LagEH} produces the equations
of \cref{EulerHeisenbergKineticOutput}. The $\rho$-weighted

$\partial_{z}^{2}\mathcal{A}_{x}$ and $6\Bzero^{2}\rho$ on
$\partial_{z}^{2}\mathcal{A}_{y}$; the $\sigma$ contribution is
restricted to the Hamiltonian density.

% --- E.3  Minimally-coupled Einstein--Cartan ------------------------
\subsection{Minimally-coupled Einstein--Cartan}%
\label{SymbolicReference}
Three quadratic torsion invariants with couplings $\alpha_{1}$,
$\alpha_{2}$, $\alpha_{3}$ are added to Einstein--Maxwell; no
curvature-squared or nonminimal terms appear. The torsion irreducible
components $\mathcal{T}_{i}$ appear only in equations with LHS $=0$.

% --- E.4  Propagating PGT with the curvature-squared term -----------
\subsection{Propagating PGT with the curvature-squared term}%
\label{SymbolicMechanism}
The $b_{5}\tilde{\mathcal{R}}^{2}$ term adds entries proportional
to $b_{5}$ with up to fourth-order time derivatives on the LHS;
the $h_{4}$, $h_{7}$, $h_{9}$ kinetic coefficients carry a
$2\,b_{5}$ factor.

\paragraph*{Observed coincidence in the $h_{\times}$ and $a_{x}$ kinetic
coefficients.}
The kinetic coefficients of the $h_{\times}$ and $a_{x}$ entries of
\cref{PropagatingKineticOutput} have been observed to coincide in
every propagating-PGT theory the pipeline has been run on:
\begin{align}
K_{h_{\times} h_{\times}} \;=\; K_{a_{x} a_{x}} .%
\label{HxAxIdentity}
\end{align}

% --- E.5  Ricci--EM nonminimal coupling -----------------------------
\subsection{Ricci--EM nonminimal coupling}%
\label{SymbolicNonminimal}
A single new vertex
$\delta_{1}\tilde{\mathcal{R}}_{\mu\nu}F^{\mu\nu}$ is added to the
Einstein--Cartan terms. The $\delta_{1}$ coupling adds entries in
every $h$-channel equation and second-time-derivative entries in
the irreducible-torsion sector.

\paragraph*{Trace-channel kinetic excerpt.}
The diagonal entries of the kinetic matrix in the irreducible-torsion
basis $(\T[1]{}, \T[2]{}, \T[3]{})$ extracted from
\cref{NonminimalKineticOutput} are
\begin{align}
K_{\T[1]{}\T[1]{}},\quad K_{\T[2]{}\T[2]{}},\quad K_{\T[3]{}\T[3]{}} .%
\label{TraceKineticOutput}
\end{align}
In \cref{TraceKineticOutput}, $K_{\T[3]{}\T[3]{}}$ has the opposite
sign of $K_{\T[1]{}\T[1]{}}$ and $K_{\T[2]{}\T[2]{}}$ for
$\delta_{1}\neq 0$.

\subsection{Full YM--PGT nonminimal}%
\label{SymbolicGeneralNonminimal}

$\beta_{1,2,3}$, $\xi$, $\delta_{1}$, $\chi$, $\zeta_{1,2,3}$
simultaneously. The Bahamonde, Barker, and Shapiro variants of
\cref{SectorSurvey} are parameter restrictions of the same

% --- E.8  Dark-photon-plasma analogue -------------------------------
\subsection{Dark-photon-plasma analogue}%
\label{SymbolicDarkPhotonPlasma}
A kinetic-mixing vertex $\delta_{m}\,F_{\mu\nu}\tilde{F}^{\mu\nu}$
(with $\tilde{F}$ the torsion-trace field strength) and a photon
mass term $M_{\mathcal{A}}^{2}$ are added to the Einstein--Cartan
Lagrangian. $M_{\mathcal{A}}^{2}$ enters the diagonal photon kinetic
term; $\delta_{m}$ enters the off-diagonal photon--torsion entries.
