% --- E.1  Einstein--Maxwell Gertsenshtein baseline ------------------
\subsection{Einstein--Maxwell Gertsenshtein baseline}%
\label{SymbolicBaseline}
The pure Einstein--Maxwell theory
$\mathcal{L} = \kappa^{-2}\mathcal{R} - \tfrac{1}{4}F_{\mu\nu}F^{\mu\nu}$
on a uniform magnetic background $\bm{B} = \Bzero\,\hat{x}$ is the
baseline against which every torsion theory below is read. The two
transverse graviton polarisations are sourced linearly by
$\Bzero\,\partial_{z}$ derivatives of the transverse photon modes;
the kinetic sub-matrix is block-diagonal in the two transverse
polarisation channels and the longitudinal photon modes decouple
from the graviton sector.

\begin{figure*}[t]
\[
\begin{aligned}
\input{sections/appendices/listings/eom_baseline_full.tex}
\end{aligned}
\]
\caption{Transverse graviton polarisations $\mathcal{H}_{xy}$ and
$\mathcal{H}_{yy}$ couple to the transverse photon modes
$\mathcal{A}_{x}$ and $\mathcal{A}_{y}$ through the $\Bzero$ vertex;
the longitudinal photon modes $\mathcal{A}_{t}$ and
$\mathcal{A}_{z}$ are free.}%
\label{BaselineKineticOutput}
\end{figure*}

% --- E.2  Euler--Heisenberg QED 1-loop -------------------------------
\subsection{Euler--Heisenberg QED corrections}%
\label{SymbolicEulerHeisenberg}
The QED 1-loop effective action
$\mathcal{L} \supset \tfrac{1}{8}\rho(F_{\mu\nu}F^{\mu\nu})^{2}
+ \tfrac{1}{8}\sigma(F_{\mu\nu}\tilde{F}^{\mu\nu})^{2}$
adds the leading parity-even and parity-odd quartic terms to
Einstein--Maxwell~\cite{heisenberg1936consequences,schwinger1951gauge,dunne2004heisenberg}.
The pipeline derives the linearised photon equations of motion via
the standard Component-E--L path and tags every $O(\rho,\sigma)$
correction for the perturbative driver (Pass 0 keeps the
$\rho=\sigma=0$ photon, Pass 1 supplies the Duhamel correction).
The polarisation-dependent dispersion shifts visible directly in
the equations -- the $2\Bzero^{2}\rho$ coefficient on the
$\mathcal{A}_{x}$ propagator versus $6\Bzero^{2}\rho$ on
$\mathcal{A}_{y}$, with the $\sigma$ term loaded onto the parallel
polarisation -- reproduce Adler's birefringence ratio
$(n_{\parallel}-1)/(n_{\perp}-1) = 4\sigma/\rho =
7/4$~\cite{adler1971photon} on the QED weak-field
coefficients~$\rho = 4\alpha^{2}/(45 m_{e}^{4}\pi^{2})$ and
$\sigma = 7\alpha^{2}/(90 m_{e}^{4}\pi^{2})$.

\begin{figure*}[t]
\[
\begin{aligned}
\input{sections/appendices/listings/eom_euler_heisenberg_full.tex}
\end{aligned}
\]
\caption{The parity-even quartic $(F^{2})^{2}$ enters every photon
mode equation with a coefficient set by $\Bzero^{2}\rho$ and
discriminates between propagation directions: the perpendicular
polarisation $\mathcal{A}_{x}$ receives a factor $2\Bzero^{2}\rho$,
the parallel polarisation $\mathcal{A}_{y}$ receives $6\Bzero^{2}\rho$.
The parity-odd $(F\tilde{F})^{2}$ term enters only through the
Hamiltonian density via the $\sigma$ coupling, where it loads
exclusively onto the parallel polarisation.}%
\label{EulerHeisenbergKineticOutput}
\end{figure*}

% --- E.3  Minimally-coupled Einstein--Cartan ------------------------
\subsection{Minimally-coupled Einstein--Cartan}%
\label{SymbolicReference}
Three quadratic torsion invariants with couplings $\alpha_{1}$,
$\alpha_{2}$, $\alpha_{3}$ are added to Einstein--Maxwell; no
curvature-squared or nonminimal terms appear. The torsion irreducible
components $\mathcal{T}_{i}$ enter only as algebraic auxiliary fields
satisfying the constraint equations whose right-hand sides vanish.

\begin{figure*}[t]
\[

\caption{Torsion irreducible components $\mathcal{T}_{i}$ enter
through algebraic constraints (the $0\!=\!\ldots$ lines); the
dynamical content of the graviton--photon sector is identical to
\cref{BaselineKineticOutput} once the constraints are solved for the
auxiliary fields.}%
\label{ReferenceKineticOutput}
\end{figure*}

% --- E.4  Propagating PGT with the curvature-squared term -----------
\subsection{Propagating PGT with the curvature-squared term}%
\label{SymbolicMechanism}
Augmenting the Einstein--Cartan Lagrangian with the parity-even
curvature-squared invariant $b_{5}\tilde{\mathcal{R}}^{2}$ promotes
selected torsion modes to propagating fields. Their equations now
carry second time-derivatives that scale linearly with $b_{5}$.

\end{figure*}

\paragraph*{Observed coincidence in the $h_{\times}$ and $a_{x}$ kinetic
coefficients.}
The kinetic coefficients of the $h_{\times}$ and $a_{x}$ entries of
\cref{PropagatingKineticOutput} coincide in the symbolic output for
every propagating-PGT theory the pipeline has been run on:
\begin{align}
K_{h_{\times} h_{\times}} \;=\; K_{a_{x} a_{x}} .%
\label{HxAxIdentity}
\end{align}

% --- E.5  Ricci--EM nonminimal coupling -----------------------------
\subsection{Ricci--EM nonminimal coupling}%
\label{SymbolicNonminimal}
A single new vertex $\delta_{1}\tilde{\mathcal{R}}_{\mu\nu}F^{\mu\nu}$
is added to the Einstein--Cartan terms. The new vertex enters every
$h$-channel equation as an off-diagonal coupling proportional to
$\delta_{1}$ and reorganises the irreducible-torsion sector through
the second-time-derivative terms it introduces.

\label{NonminimalKineticOutput}
\end{figure*}

\paragraph*{Trace-channel kinetic excerpt.}
The diagonal entries of the kinetic matrix in the irreducible-torsion
basis $(\T[1]{}, \T[2]{}, \T[3]{})$ extracted from
\cref{NonminimalKineticOutput} are
\begin{align}
K_{\T[1]{}\T[1]{}},\quad K_{\T[2]{}\T[2]{}},\quad K_{\T[3]{}\T[3]{}} ,%
\label{TraceKineticOutput}
\end{align}
and the sign of each entry is what the no-ghost analysis below reads
off.

\paragraph*{Sign-positivity of the irreducible torsion kinetic
matrices.}
The no-ghost condition on the irreducible-torsion sector is the
requirement that every diagonal entry of the kinetic matrix in the
$(\T[1]{},\T[2]{},\T[3]{})$ basis be positive in the metric signature
of \cref{Linearisation}; the four-parameter classification of
\citet{sezgin1980ghostfree} and its modern PGT refinements
\cite{lin2019pgt,barker2021hamiltonian} express the same condition as
algebraic inequalities on the Lagrangian couplings. Read against
\cref{TraceKineticOutput}, the surveyed Ricci--EM nonminimal theory
satisfies the no-ghost condition in the $\T[1]{}$ and $\T[2]{}$
sectors but violates it in the trace $\T[3]{}$ sector for
$\delta_{1}\neq 0$; the kinematic observation that this wrong-sign
sector does not couple to the $h_{\times} \leftrightarrow a_{x}$
Gertsenshtein channel is recorded in \cref{ChannelStructure}.

% --- E.6  Full YM--PGT nonminimal -----------------------------------
\subsection{Full YM--PGT nonminimal}%
\label{SymbolicGeneralNonminimal}
The full Yang--Mills-form Poincar\'e gauge theory turns on all
nonminimal parity-even couplings -- $\beta_{1,2,3}$, $\xi$,
$\delta_{1}$, $\chi$, $\zeta_{1,2,3}$ -- and is the broadest
Lagrangian swept in the campaign of \cref{Results}. The Bahamonde,
Barker, and Shapiro variants surveyed in \cref{SectorSurvey} are
parameter restrictions of this Lagrangian; their kinetic structure
is read off the same block under the corresponding substitutions.

\subsection{Parity-odd extension}%
\label{SymbolicParityOdd}
The parity-odd extension adds twelve parity-odd invariants
($d_{14}$--$d_{17}$, $d_{19}$--$d_{21}$, $\zeta_{T,1}$--$\zeta_{T,6}$
in the notation of \cref{SectorSurvey}) on top of the full
Yang--Mills-form Lagrangian. These vertices are the leading
P-violating tree-level corrections to PGT that the campaign
constrains.

% --- E.8  Dark-photon-plasma analogue -------------------------------
\subsection{Dark-photon-plasma analogue}%
\label{SymbolicDarkPhotonPlasma}
Promoting the torsion trace to the role of a Proca-mass dark photon
proceeds by adding a kinetic-mixing vertex
$\delta_{m}\,F_{\mu\nu}\tilde{F}^{\mu\nu}$ on top of the
Einstein--Cartan Lagrangian, where $\tilde{F}$ here denotes the
torsion-trace field strength rather than the Hodge dual. The photon
mass $M_{\mathcal{A}}^{2}$ enters the diagonal photon kinetic term;
the kinetic mixing $\delta_{m}$ enters every photon EOM as an
off-diagonal coupling to the irreducible-torsion sector.
