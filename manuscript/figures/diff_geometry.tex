% Three geometric obstructions (curvature, torsion, non-metricity)
% schematic figure. Adapted from figure 1 of Barker, Marzo & Santoni,
% "Can metric-affine gravity be saved?", Phys. Rev. D 112 044032 (2025),
% arXiv:2505.23894 — bib key barker2025metricaffine.
\documentclass[border=3pt,tikz,svgnames]{standalone}
\usepackage{stix} % match the body-document font for visual consistency
\usepackage{tikz}
\usetikzlibrary{decorations.markings,arrows.meta,calc}
\tikzset{>={Stealth[length=1.2mm,width=1.2mm]}} % default arrow tip for -> etc.
\colorlet{OrangeFill}{LightSalmon}
\colorlet{OrangeLine}{DarkSalmon!30!Chocolate}
\colorlet{GreenFill}{MediumAquamarine}
\colorlet{GreenLine}{MediumAquamarine!30!Teal}
\colorlet{PurpleFill}{MediumPurple}
\colorlet{PurpleLine}{MediumPurple!30!SlateBlue}

\tikzset{
  DashedArrows/.style 2 args={
    densely dashed, thin,
    postaction={decorate},
    decoration={markings,
      mark=at position #1 with {\arrow{>}},
      mark=at position #2 with {\arrow{>}}
    }
  }
}

\begin{document}

\begin{tikzpicture}[line cap=round]
    \def\Lx{1.44}
    \def\Ly{0.96}
    \begin{scope}[shift={(0,0)}]
        \path[use as bounding box] (-\Lx,-\Ly) rectangle (\Lx,\Ly);
        \def\Rx{0.7*\Lx}
        \def\Ry{0.45*\Ly}
        \coordinate (O) at (0,-0.2*\Ly);
        \draw[DashedArrows={0.17}{0.62}] (0,-0.2*\Ly) ellipse ({\Rx} and {\Ry});
        \fill[OrangeFill, opacity=0.4] (\Rx,0.3*\Ly) arc(0:-90:{\Rx} and {\Ry}) --++ (0,-0.5*\Ly) arc(-90:0:{\Rx} and {\Ry}) -- cycle;
        \fill[OrangeFill, opacity=0.4] (\Rx,0.3*\Ly) arc(0:180:{1.05*\Rx} and {0.95*\Ry}) -- (-1.1*\Rx,0.3*\Ly) to[out=-90,in=103] (-\Rx,-0.2*\Ly) arc(180:0:{\Rx} and {\Ry}) -- cycle;
        \fill[OrangeFill, opacity=0.4] ($(O)+(0,-\Ry)$) arc(-90:-180:{\Rx} and {\Ry}) to[out=103,in=-90] (-1.1*\Rx,0.3*\Ly) to[out=-90,in=180] ($(O)+(0,-\Ry)+(115:0.4*\Ly)$) -- cycle;

        \draw[OrangeLine, DashedArrows={0.17}{0.62}] ($(O)+(0,0.5*\Ly-\Ry)$) arc(-90:0:{\Rx} and {\Ry}) arc(0:180:{1.05*\Rx} and {0.95*\Ry}) to[out=-90,in=180] ($(O)+(0,-\Ry)+(115:0.4*\Ly)$);

        \draw[OrangeLine, thick, ->] ($(O)+(0,-\Ry)$) --++ (0,0.5*\Ly);
        \draw[OrangeLine, thick, ->] ($(O)+(0,-\Ry)$) --++ (115:0.4*\Ly);

        \node at (0,0) {$\mathcal{F}$};
    \end{scope}

    \begin{scope}[shift={(2*\Lx,0)}]
        \path[use as bounding box] (-\Lx,-\Ly) rectangle (\Lx,\Ly);
        \coordinate (O) at (-0.487*\Lx,-0.63*\Ly);
        \coordinate (R) at ($(O)+(98:0.9*\Lx)$);
        \coordinate (Ru) at ($(O)+(98:0.9*\Lx)+(10:0.7*\Lx)$);
        \coordinate (Q) at ($(O)+(5:0.9*\Lx)$);
        \coordinate (Qv) at ($(O)+(5:0.9*\Lx)+(85:0.7*\Lx)$);

        \draw[PurpleLine, thick, ->] (O) -- (R);
        \draw[PurpleLine, thick, ->] (O) -- (Q);
        \draw[PurpleLine, thick, ->] (R) -- (Ru);
        \draw[PurpleLine, thick, ->] (Q) -- (Qv);
        \draw[PurpleLine, dashed, thin] (Q) to[bend right=15] (Ru);
        \draw[PurpleLine, dashed, thin] (R) to[bend left=15] (Qv);

        \fill[PurpleFill, opacity=0.2] (O) -- (Q) to[bend right=15] (Ru) -- (R) -- cycle;
        \fill[PurpleFill, opacity=0.2] (O) -- (R) to[bend left=15] (Qv) -- (Q) -- cycle;

        \draw[densely dashed, thin,->] (-0.45*\Lx,-0.95*\Ly) to[bend left=12] (-0.43*\Lx,0.99*\Ly);
        \draw[densely dashed, thin,->] (-0.8*\Lx,-0.7*\Ly) to[bend left=15] (0.8*\Lx,-0.9*\Ly);

        \node at (0,0) {$\mathcal{T}$};
    \end{scope}

    \begin{scope}[shift={(4*\Lx,0)}]
        \path[use as bounding box] (-\Lx,-\Ly) rectangle (\Lx,\Ly);
        \coordinate (A) at (-0.8*\Lx,0.66*\Ly);
        \coordinate (B) at (-0.61*\Lx,0.51*\Ly);
        \coordinate (C) at (-0.4*\Lx,0.46*\Ly);
        \coordinate (D) at (0.37*\Lx,0.26*\Ly);
        \coordinate (E) at (0.58*\Lx,0.50*\Ly);
        \coordinate (F) at (0.8*\Lx,0.76*\Ly);

        \coordinate (G) at ($(E)+(-75:0.7*\Lx)$);
        \coordinate (H) at (0.65*\Lx,-0.34*\Ly);
        \coordinate (I) at (0.54*\Lx,-0.17*\Ly);
        \coordinate (J) at (0.37*\Lx,-0.22*\Ly);
        \coordinate (K) at (0.2*\Lx,-0.54*\Ly);
        \coordinate (L) at (-0.05*\Lx,-0.44*\Ly);
        \coordinate (M) at (-0.3*\Lx,-0.64*\Ly);
        \coordinate (N) at ($(B)+(-75:\Lx)$);

        \fill[GreenFill, opacity=0.3] (N) -- (B) to[out=-20, in=170] (C) to[out=-10, in=-160] (D) to[out=20, in=-120] (E) -- (G) to[out=-170, in=-70] (H) to[out=110, in=-60] (I) to[out=120, in=100] (J) to[out=-80, in=10] (K) to[out=-170, in=-60] (L) to[out=120, in=105] (M) to[out=-85, in=20] (N);

        \draw[GreenLine, thick, ->] (B) --++(-75:\Lx);
        \draw[GreenLine, thick, ->] (E) -- (G);

        \draw[GreenLine, dashed, thin] (G) to[out=-170, in=-70] (H) to[out=110, in=-60] (I) to[out=120, in=100] (J) to[out=-80, in=10] (K) to[out=-170, in=-60] (L) to[out=120, in=105] (M) to[out=-85, in=20] (N);

        \draw[DashedArrows={0.22}{0.75}] (A) to[out=-35, in=160] (B) to[out=-20, in=170] (C) to[out=-10, in=-160] (D) to[out=20, in=-120] (E) to[out=60, in=-160] (F);

        \node at (0,0) {$\mathcal{Q}$};
    \end{scope}
\end{tikzpicture}

\end{document}
