    {\scriptsize $d$ = dynamical,\quad $c$ = constraint}\\[2pt]

    \parbox{5.4cm}{\centering
    \textbf{Schur complement}\\[4pt]

    {\scriptsize field correction: $A_{dc,\mathrm{field}} \cdot \mathrm{recovery}$}\\[2pt]
    {\scriptsize velocity coupling (implicit):}\\[1pt]
    {\scriptsize $\dot{d} = (I - A_{dc,v}\cdot\mathrm{recovery})^{-1}\,A_{\mathrm{rhs}}\,d$}%
    }};

% ===================================================================
% REDUCED SYSTEM
% ===================================================================
\node[stage=Green, below=of schur] (reduced) {%
    \textbf{Reduced system} (dynamical only)\\[2pt]
    \scriptsize $\dot{d} = A_{\mathrm{red}}(k)\,d$ \quad---\quad eigendecompose as usual};
