\begin{tikzpicture}[
    node distance=0.8cm
]

% ===================================================================
% ELIGIBILITY GATE (top)
% ===================================================================
\node[check, text width=3.2cm] (gate)
    {\textbf{\texttt{can\_use\_modal()}}\\[2pt]

     supported operators\\
     time-independent coeffs\\
     constraints eliminable};

% ===================================================================
% PHYSICAL SPACE INPUT
% ===================================================================
\node[stage=Gray,
      left=0.8cm of gate.west] (input)
    {$\mathbf{y}(x, t_0)$\\[-1pt]\scriptsize physical space};

% ===================================================================
% FFT
% ===================================================================
\node[stage=Purple,
      below=0.4cm of input] (fft)
    {\textbf{FFT}\\[-1pt]\scriptsize per field slot};

% ===================================================================
% DECISION: Position-dependent coefficients?
% ===================================================================
\node[decision, below=0.4cm of fft] (branch)
    {Position-dep.\\coefficients?};

% ===================================================================
% PATH A: Per-mode eigendecomposition (upper branch, purple)
% ===================================================================
\node[stage=Pink,
      below right=0.8cm and 0.6cm of branch.east, anchor=north] (matrix_pm)
    {Build $A(k)$\\[-1pt]\scriptsize independent per mode $k$};

\node[stage=Pink,
      below=0.4cm of matrix_pm] (eigen)
    {Eigendecompose\\[-1pt]\scriptsize $A = V \Lambda V^{-1}$};

\node[stage=Pink,
      below=0.4cm of eigen] (evolve_eig)
    {Evolve\\[-1pt]\scriptsize $e^{\Lambda \Delta t}$};

% ===================================================================
% PATH B: expm_multiply (lower branch, lime)
% ===================================================================
\node[stage=Blue,
      below left=0.8cm and 0.6cm of branch.west, anchor=north] (matrix_full)
    {Build full matrix\\[-1pt]\scriptsize $c(x)\!\cdot\!u \to \hat{c}\ast\hat{u}$ couples all modes};

\node[stage=Blue,
      below=0.4cm of matrix_full] (sparsify)
    {Sparsify\\[-1pt]\scriptsize zero negligible entries; sparse CSC};

\node[stage=Blue,
      below=0.4cm of sparsify] (expm)
    {\texttt{expm\_multiply}\\[-1pt]\scriptsize $\exp(A \cdot t) \cdot y_0$ (all $t$ at once)};

% ===================================================================
% CONVERGENCE: IFFT + output
% ===================================================================
\node[stage=Purple,
      below=4.2cm of branch] (ifft)
    {\textbf{IFFT}\\[-1pt]\scriptsize per field slot};

\node[stage=Gray,
      below=0.4cm of ifft] (output)
    {$\mathbf{y}(x, t)$\\[-1pt]\scriptsize physical space};

% ===================================================================
% ARROWS — main flow
% ===================================================================
\draw[arr, densely dotted, OrangeLine] (gate.west) -- (input.east);
\draw[arr] (input) -- (fft);
\draw[arr] (fft) -- (branch);

% Path A (upper): No position-dependent → per-mode eigendecomp
\draw[arr] (branch.east) -| node[yeslbl, below left] {No} (matrix_pm.north);
\draw[arr] (matrix_pm) -- (eigen);
\draw[arr] (eigen) -- (evolve_eig);
\draw[arr, rounded corners=3pt]
    (evolve_eig.south) |- (ifft.east);

% Path B (lower): Yes position-dependent → sparsify → expm_multiply
\draw[arr] (branch.west) -| node[nolbl, below right] {Yes} (matrix_full.north);
\draw[arr] (matrix_full) -- (sparsify);
\draw[arr] (sparsify) -- (expm);
\draw[arr, rounded corners=3pt]
    (expm.south) |- (ifft.west);

% IFFT -> output
\draw[arr] (ifft) -- (output);


% ===================================================================
% BLOCK-AWARE CHECK (above per-mode matrix, Path A only)
% ===================================================================
\node[check, above =1.2cm of matrix_pm.north east, text width=3.4cm] (blocks)
    {Detect decoupled field blocks\\[1pt]
     skip blocks with zero initial data\\
     warn on unstable eigenvalues};
\draw[arr, densely dotted, OrangeLine] (blocks.south) -- (matrix_pm.north east);

% ===================================================================
% CONSTRAINT ELIMINATION (Schur complement, Path A only)
% ===================================================================
\node[detail=Pink, right=0.4cm of matrix_pm] (schur)
    {\textbf{Schur complement}\\[2pt]
     eliminate constraints:\\
     $A_{\mathrm{red}} = A_{dd} - A_{dc} S_{cc}^{-1} S_{cd}$\\[2pt]
     (dynamical DOFs only)};

\node[annot, above=0.1cm of schur] {if constraints present};

% Arrow from matrix_pm down to Schur, then to eigen
\draw[arr, PinkLine, densely dashed, rounded corners=3pt]
    (matrix_pm.east) -- (schur.west);
\draw[arr, PinkLine, densely dashed, rounded corners=3pt]
    (schur.south) |- (eigen.east);

% Recovery arrow from evolve_eig down to reconstruction
\node[detail=Pink, below right=0.2cm and -1.2cm of evolve_eig, text width=2.8cm] (recover)
    {\textbf{Reconstruct}\\[2pt]
     $\hat{c}(k) = -S_{cc}^{-1} S_{cd}\, \hat{d}(k)$\\[2pt]
     IFFT $\to c(x,t)$};
\draw[arr, PinkLine, densely dashed, rounded corners=3pt]
    (evolve_eig.east) -| (recover.north);
\draw[arr, PinkLine, densely dashed, rounded corners=3pt]
    (recover.south) |- (ifft.south east);

% ===================================================================
% BACKGROUND ZONES

    \node[fit=(input)(output), inner sep=12pt, fill=GrayFill, densely dashed, draw=GrayLine, fill opacity=0.12, rounded corners=5pt] {};
    % Fourier space zone (purple tint) — encompasses both branches
    \node[fit=(fft)(ifft)(matrix_pm)(evolve_eig)(matrix_full)(sparsify)(expm),
          inner sep=4pt, fill=PurpleFill, draw=PurpleLine, densely dashed,
          fill opacity=0.12, rounded corners=5pt] {};
\end{scope}

% Zone labels
\node[font=\scriptsize\bfseries, text=GrayLine,
      above left=0.0cm and -0.4cm of input.north, anchor=south east]
    {Physical space};
\node[font=\scriptsize\bfseries, text=PurpleLine,
      above left=0.2cm and 0.8cm of fft.west, anchor=east]
    {Fourier space};

% ===================================================================
% PATH LEGEND (bottom right)
% ===================================================================
\node[font=\scriptsize, align=left,
      below right=0.2cm and 0.8cm of output.east, anchor=west,
      draw=GrayLine, rounded corners=2pt, inner sep=5pt, fill=white] {
    \textcolor{PinkLine}{\rule{0.5cm}{2pt}}

    \textcolor{BlueLine}{\rule{0.5cm}{2pt}}

    \textcolor{PinkLine}{\rule[1pt]{0.5cm}{1pt}}

};
