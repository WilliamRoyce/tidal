\documentclass[border=10pt]{standalone}
\usepackage{tidal-tikz-styles}

\begin{document}
\begin{tikzpicture}[
    node distance=0.55cm
]

% Stages
\node[stage=Gray] (toml)
    {\textbf{Configuration}\\[1pt]
     \scriptsize covariant components: $\bar{A}_y = -B_0 z$};

\node[stage=Pink, below=of toml] (declare)
    {Declare background tensor $\bar{A}_\mu$\\[1pt]
     \scriptsize in computer algebra system};

\node[stage=Pink, below=of declare] (assign)
    {Assign component values\\[1pt]
     \scriptsize auto-evaluating substitution};
\node[annot, right=0.3cm of assign.east, anchor=west]
    {Components collapse immediately\\during symbolic expansion};

\node[stage=Purple, below=of assign] (deriv)
    {Evaluate coordinate derivatives $\partial_\mu \bar{A}_\nu$};
\node[annot, right=0.3cm of deriv.east, anchor=west]
    {Applied before + after\\
     conversion to partial derivatives};

\node[stage=Purple, below=of deriv] (fstrength)
    {Evaluate background field strength\\[1pt]
     \scriptsize $\bar{F}_{xy} = \partial_x \bar{A}_y - \partial_y \bar{A}_x = B_0$};

\node[stage=Green, below=of fstrength] (export)
    {\textbf{Export position-dependent coefficients}};

\node[stage=Blue, below=of export] (pyeval)
    {\textbf{Evaluate coefficients on spatial grid}\\[1pt]
     \scriptsize overflow-safe ($1/e^{x^2} \to e^{-x^2}$)};

\node[stage=Blue, below=of pyeval] (cache)
    {\textbf{Cache on grid} (evaluated once)};
\node[annot, right=0.3cm of cache.east, anchor=west]
    {Position-dependent but\\time-independent\\$\to$ single evaluation};

% Arrows
\foreach \a/\b in {toml/declare, declare/assign, assign/deriv,
                   deriv/fstrength, fstrength/export, export/pyeval, pyeval/cache} {
    \draw[arr] (\a) -- (\b);
}

\end{tikzpicture}
