\documentclass[border=10pt]{standalone}
\usepackage{tidal-tikz-styles}

\newcommand{\costlbl}[2]{{\scriptsize\bfseries\color{#1}#2}}

\begin{document}
\begin{tikzpicture}[
    costlbl/.style={font=\scriptsize\bfseries, text=#1},
    node distance=0.4cm
]

% ===================================================================
% TITLE
% ===================================================================
\node[font=\normalsize\bfseries] (title) at (0,0)
    {Coefficient Classification and Evaluation Strategy};

% ===================================================================
% LEVELS (top = fastest/most common, bottom = slowest/rarest)
% ===================================================================

% L0
\node[stage=Green, below=0.6cm of title, text width=9.6cm] (l0) {%
    \textbf{L0 --- Pre-resolved constants} \hfill
    \costlbl{GreenLine}{$O(1)$ lookup}\\[2pt]
    Literal numeric values (e.g., $1.0$, $-0.5$).
    Substituted at initialisation.\\
    \textit{Example:} coefficient $= 1.0$ (no symbolic expression)};

% L1
\node[stage=Blue, below=of l0, text width=9.6cm] (l1) {%
    \textbf{L1 --- Named parameters} \hfill
    \costlbl{BlueLine}{$O(1)$ eval + lookup}\\[2pt]
    Symbolic expressions resolved via input parameters
    (e.g., mass $m^2$, coupling $g$). Evaluated once.\\
    \textit{Example:} $m_A^2 \to 1.0$ from input specification};

% L2
\node[stage=Purple, below=of l1, text width=9.6cm] (l2) {%
    \textbf{L2 --- Spatially-varying coefficients} $c(\mathbf{x})$ \hfill
    \costlbl{PurpleLine}{$O(N)$ eval, cached}\\[2pt]
    Position-dependent coefficients evaluated on the spatial grid once.\\
    \textit{Example:} $c = B_0 \cdot x$ (depends on spatial coordinate)};

% L3
\node[stage=Orange, below=of l2, text width=9.6cm] (l3) {%
    \textbf{L3 --- Time-dependent coefficients} $c(\mathbf{x}, t)$ \hfill
    \costlbl{OrangeLine}{$O(N)$ eval per step, deduped}\\[2pt]
    Re-evaluated each timestep.
    Deduplicated across Newton iterations within a step.\\
    \textit{Example:} $c = e^{-\gamma t}$ (exponential decay)};

% ===================================================================
% ARROWS (fallthrough)
% ===================================================================
\draw[arr] (l0.south west) ++(0.3,0) -- ++(0,-0.35)
    node[midway, right, font=\scriptsize, text=GrayLine] {miss};
\draw[arr] (l1.south west) ++(0.3,0) -- ++(0,-0.35)
    node[midway, right, font=\scriptsize, text=GrayLine] {miss};
\draw[arr] (l2.south west) ++(0.3,0) -- ++(0,-0.35)
    node[midway, right, font=\scriptsize, text=GrayLine] {miss};

% Hit labels
\node[costlbl=GreenLine, right=0.3cm of l0.east] {hit $\to$ return};
\node[costlbl=BlueLine, right=0.3cm of l1.east] {hit $\to$ return};
\node[costlbl=PurpleLine, right=0.3cm of l2.east] {hit $\to$ return};
\node[costlbl=OrangeLine, right=0.3cm of l3.east] {hit $\to$ return};

% ===================================================================
% FREQUENCY ANNOTATION (left brace)
% ===================================================================
\draw[decorate, decoration={brace, amplitude=6pt, raise=2pt, mirror},
      GreenLine, thick]
    (l0.north west) -- (l0.south west)
    node[midway, left=10pt, font=\scriptsize, text=GreenLine, align=right]
    {most\\common};

\draw[decorate, decoration={brace, amplitude=6pt, raise=2pt, mirror},
      OrangeLine, thick]
    (l3.north west) -- (l3.south west)
    node[midway, left=10pt, font=\scriptsize, text=OrangeLine, align=right]
    {rarest};

\end{tikzpicture}
\end{document}
