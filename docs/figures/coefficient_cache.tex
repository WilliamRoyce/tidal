% ===================================================================
% TITLE
% ===================================================================
\node[font=\normalsize\bfseries] (title) at (0,0)
    {Coefficient Resolution Hierarchy (4-Level Cache)};

\node[annot, below=0.1cm of title]  (sub)
    {\texttt{CoefficientEvaluator} in \texttt{coefficients.py}};

% ===================================================================
% LEVELS (top = fastest/most common, bottom = slowest/rarest)
% ===================================================================

% L0
\node[level=GreenLine, fill=GreenFill, fill opacity=0.2,
      below=0.6cm of sub, text width=9.6cm] (l0) {%
    \textbf{L0 --- Pre-resolved constants} \hfill
    \costlbl{GreenLine}{$O(1)$ dict lookup}\\[2pt]
    Numeric coefficients with no symbolic expression (e.g., \texttt{1.0}, \texttt{-0.5}).\\
    Resolved at \texttt{\_\_init\_\_()} and stored in \texttt{\_constants} dict.\\
    \textit{Example:} $\texttt{coefficient} = 1.0$, $\texttt{coefficient\_symbolic} = \texttt{None}$};

% L1
\node[level=BlueLine, fill=BlueFill, fill opacity=0.15,
      below=of l0, text width=9.6cm] (l1) {%
    \textbf{L1 --- Parametric constants} \hfill
    \costlbl{BlueLine}{$O(1)$ eval + param lookup}\\[2pt]
    Symbolic expressions resolved via \texttt{--param} overrides (e.g., \texttt{"m2"} $\to$ 1.0).\\
    Mathematica$\to$Python string conversion cached once.\\
    \textit{Example:} $\texttt{coefficient\_symbolic} = \texttt{"mA2"}$, resolved to \texttt{float}};

% L2
\node[level=PurpleLine, fill=PurpleFill, fill opacity=0.12,
      below=of l1, text width=9.6cm] (l2) {%
    \textbf{L2 --- Spatial grid arrays} \hfill
    \costlbl{PurpleLine}{$O(N)$ eval, cached}\\[2pt]
    Position-dependent coefficients evaluated on the spatial grid once.\\
    Stored in \texttt{\_spatial\_cache[(eq\_idx, term\_idx)]} as \texttt{ndarray}.\\
    \textit{Example:} $\texttt{coefficient\_symbolic} = \texttt{"B0*x[]"}$,
    $\texttt{coordinate\_dependent} = \texttt{["x"]}$};

% L3
\node[level=OrangeLine, fill=OrangeFill, fill opacity=0.12,
      below=of l2, text width=9.6cm] (l3) {%
    \textbf{L3 --- Per-timestep dedup} \hfill
    \costlbl{OrangeLine}{$O(N)$ eval per step, deduped}\\[2pt]
    Time \emph{and} position-dependent coefficients re-evaluated each timestep.\\
    Cached within one step to deduplicate across Newton iterations (IDA).\\
    Cleared by \texttt{begin\_timestep()}.\\
    \textit{Example:} $\texttt{coefficient\_symbolic} = \texttt{"Exp[-dSH*t[]]"}$,
    $\texttt{time\_dependent} = \texttt{true}$};

% ===================================================================

\node[costlbl=OrangeLine, right=0.3cm of l3.east] {hit $\to$ return};

    {most\\common};

      OrangeLine, thick]
    (l3.north west) -- (l3.south west)
    node[midway, left=10pt, font=\scriptsize, text=OrangeLine, align=right]
    {rarest};

% ===================================================================
% VALIDATION CHECKS (below)
% ===================================================================
\node[check, below=0.6cm of l3, text width=9.6cm] (checks) {%
    \textbf{Init-time checks} (\texttt{\_\_init\_\_()}):\\[2pt]
    \textbullet\ \texttt{\_validate\_unresolved()}: fail-fast if any symbolic coefficient
    cannot be evaluated (missing \texttt{--param})\\
    \textbullet\ \texttt{\_check\_mass\_sign()}: diagnostic warning for unexpected sign in mass terms\\
    \textbullet\ \texttt{all\_time\_independent()}: if true, skip L3 cache clear each step
    (common-case optimisation)};
