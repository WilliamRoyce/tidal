    {\textbf{Lagrangian construction}\\[-1pt]
     \scriptsize constants, gauge terms};

    {Euler--Lagrange variation\\[-1pt]

\node[stage=Purple, below=of christoffel] (replace)

    {Covariant $\to$ partial derivatives\\[-1pt]
     \scriptsize $\nabla_a \to \partial_t, \partial_x, \ldots$};

\node[stage=Green, below=of convert] (export)
    {\textbf{Export to intermediate representation}\\[-1pt]
     \scriptsize coefficients, operators, fields};

    {\textbf{\texttt{spec.json}}};
