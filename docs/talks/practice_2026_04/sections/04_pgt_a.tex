\begin{frame}{Torsion via Poincar\'e gauging (I)}
  \note{\textasciitilde 5.5 min mark. Frame torsion as the \emph{other half} of a gauge story they already believe in.}

  \textbf{Question.} GR uses the Levi-Civita connection (metric compatible, torsion-free). Why \emph{that} choice?

  \vspace{0.4em}
  \textbf{Utiyama (1956), Kibble (1961), Sciama (1962):} gravity as a gauge theory of the Poincar\'e group.
  \begin{itemize}
    \item QED $\leftrightarrow$ gauging $U(1)$.
    \item Yang--Mills $\leftrightarrow$ gauging $SU(N)$.
    \item Gravity $\leftrightarrow$ gauging the Poincar\'e group = translations $\ltimes$ Lorentz.
  \end{itemize}

  \vspace{0.4em}
  \textbf{Gauging both subgroups:}
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Gauge subgroup & Connection & Field strength \\
      \midrule
      Lorentz      & spin connection $\omega^{ab}{}_\mu$ & \textbf{curvature} $R^{ab}{}_{\mu\nu}$ \\
      Translations & tetrad $e^a{}_\mu$                  & \textbf{torsion}   $T^a{}_{\mu\nu}$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.3em}
  \emph{GR keeps one of the two field strengths of the Poincar\'e group. PGT keeps both.}
\end{frame}
