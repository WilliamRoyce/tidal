\begin{frame}{Torsion via Poincar\'e gauging (I)}
  \note{\textasciitilde 5.5 min mark. Frame torsion as the \emph{other half} of a gauge story they already believe in.}

  \textbf{Why gauge theory here:} local Lorentz freedom and coordinate freedom are \emph{gauge redundancies} of the Poincar\'e group --- suggesting a gauge treatment of gravity.

  \vspace{0.3em}
  \textbf{Utiyama (1956), Kibble (1961), Sciama (1962):} gauging each subgroup
  introduces a connection and a field strength:
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Gauge group & Connection & Field strength \\
      \midrule
      $U(1)$ (QED)               & $A_\mu$             & $F_{\mu\nu}$ \\
      $SO(1,3)$ (Lorentz)        & $\omega^{ab}{}_\mu$ & \textbf{curvature} $R^{ab}{}_{\mu\nu}$ \\
      $\mathbb{R}^4$ (translations) & $e^a{}_\mu$      & \textbf{torsion}   $T^a{}_{\mu\nu}$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{0.25em}
  \emph{GR ($V_4$): $\mathcal{R}\neq 0$, $\mathcal{T}=0$, $\mathcal{Q}=0$.\quad
        PGT ($U_4$): $\mathcal{R}\neq 0$, $\mathcal{T}\neq 0$, $\mathcal{Q}=0$.\quad
        Torsion is the new degree of freedom.}
\end{frame}
