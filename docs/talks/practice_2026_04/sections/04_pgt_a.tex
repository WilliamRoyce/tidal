\begin{frame}{Poincar\'e gauge theory and torsion}

  \begin{itemize}
    \item A \textbf{local} symmetry transforms \emph{independently at each spacetime point}.
    \item \emph{Connection} field restores covariance after varying the transformation parameter.
    \item Its field strength is the \emph{new physical content}.
    \item Spacetime has two such symmetries: \emph{local \textbf{Lorentz}} and \emph{local \textbf{translations}}.
  \end{itemize}
