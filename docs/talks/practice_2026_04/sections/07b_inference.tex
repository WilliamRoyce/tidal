\begin{frame}{Nested sampling: surveying the amplification landscape}
  For each parameter point $\theta$, we run a full PDE simulation and compute the amplification factor:

  \vspace{0.4em}
  \[
    L(\theta) = A(\theta)
    = \frac{P_{\max}(\theta)}{\sin^2\!\bigl(\tfrac{\kappa B_0 t}{2}\bigr)}
  \]
  We then define a score function, which we call the `likelihood': $L(\theta)=\pm \operatorname{log} A(\theta)$.

  \begin{itemize}
    \item valid in the linearised, perturbative regime: $P_{\max}\ll 1$,\;
      $\kappa B_0 t\ll 1$;\;
      tachyonic and non-perturbative (Hwang--Noh) regions $\to L=0$

    \item Central quantity: $\displaystyle\mathcal{Z}=\int A(\theta)\,\pi(\theta)\,\mathrm{d}\theta$
      \;---\; prior-averaged amplification:\\
      $\log\mathcal{Z}\approx 0$ (null: model is GR-like across coupling space)
      \quad vs \quad
      $\log\mathcal{Z} > 0$ (enhancement is generic)

    \item Divergence from prior to posterior: $D_{\mathrm{KL}}$ --- which parameters drive the signal?
  \end{itemize}
\end{frame}
