\begin{frame}{The question and the program}
  \note{\textasciitilde 7.5 min mark. Sharpest framing, then pivot to the tool.}
  \begin{block}{}
    \centering
    \emph{Is there any combination of quadratic curvature and torsion modifications whose linearised PDEs, in a background magnetic field, give an amplified Gertsenshtein conversion --- and are the surviving theories physically viable?}
  \end{block}
  \vspace{0.6em}
  \textbf{For every Lagrangian in the scan, three things to do:}
  \begin{enumerate}
    \item Derive the linearised field equations symbolically.
    \item Screen for causality-violating regimes via all-mode eigenvalue stability.
    \item Compute the conversion probability and compare to the GR baseline.
  \end{enumerate}
  \vspace{0.6em}
  All three are what \textsc{tidal} automates.
\end{frame}
