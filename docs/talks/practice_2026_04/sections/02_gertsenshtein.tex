\begin{frame}{The Gertsenshtein effect}
  \note{\textasciitilde 2.5 min mark. Audience has probably not seen this. Lead with the physical origin --- F\^{}2 has two metric factors --- then give the formula.}
  \textbf{Setup.} Linearised GR $+$ Maxwell $+$ uniform background $B_0$.

  \vspace{0.3em}
  \textbf{Why they mix.} The EM action $\propto F_{\mu\nu}F^{\mu\nu}$ contains \emph{two} inverse metric factors. Perturbing the metric around a $B_0$ background generates a cross-term between $h_{\mu\nu}$ and $\delta A$. No $B_0$, no cross-term, no mixing.

  \vspace{0.3em}
  \textbf{Result:}
  \[
    P_{g\to\gamma}(D) = \sin^{2}\!\!\left(\tfrac{1}{2}\kappa B_0 D\right)
  \]

  \begin{itemize}\small
    \item Same structure as two-state QM / axion--photon conversion (Raffelt \& Stodolsky 1988).
    \item Background $B_0$ plays the role of the off-diagonal coupling.
  \end{itemize}
\end{frame}
