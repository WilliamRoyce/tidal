\begin{frame}{The Gertsenshtein effect}
  \textbf{Setup:} Linearised GR $+$ Maxwell $+$ uniform background $B$.

  \vspace{0.3em}
  \textbf{Why they mix:} EM action $-\frac{1}{4} F_{\mu\nu} F^{\mu\nu}$ has \textcolor{red}{two metric factors} AND two EM field strengths.  Linearise $g = \eta + \textcolor{red}{h}$ and $F = \textcolor{magenta}{\bar F} + \textcolor{blue}{\delta F}$:
  \[
    S_{\rm EM} \propto \int (\eta + \textcolor{red}{h})(\eta + \textcolor{red}{h})\,(\textcolor{magenta}{\bar F} + \textcolor{blue}{\delta F})(\textcolor{magenta}{\bar F} + \textcolor{blue}{\delta F})
  \]
  Pick \emph{one $\textcolor{red}{h}$, one $\textcolor{blue}{\delta F}$, one $\textcolor{magenta}{\bar F}$}: a direct $h \leftrightarrow \gamma$ coupling $\propto \textcolor{magenta}{\bar F} = B$.

  \vspace{0.3em}
  \textbf{Result:}
  {\Large\[
    P_{g\to\gamma}(L) = \sin^{2}\! \left(\tfrac{1}{2}\kappa B L\right)
  \]}
  Two-state mixing; off-diagonal coupling $B$.

  
\end{frame}
