\begin{frame}{The Gertsenshtein effect}
  \note{\textasciitilde 2.5 min mark. Audience has probably not seen this. Introduce from scratch.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \textbf{Setup.} Linearised GR $+$ Maxwell $+$ background magnetic field $B_0$.
      \vspace{0.5em}

      \textbf{Mechanism.} The GW perturbation $h_{\mu\nu}$ couples to $F^{\mu\nu}$ through the metric; in a background $B_0$ this sources a new EM mode.
      \vspace{0.5em}

      \textbf{Result (Gertsenshtein 1962):}
      \[
        P_{g\to\gamma}(D) \;=\; \sin^{2}\!\left(\tfrac{1}{2}\kappa B_0 D\right)
      \]
      \begin{itemize}\small
        \item Amplitude $\propto B_0$, oscillation length $\propto 1/(\kappa B_0)$.
        \item Same math as two-state QM mixing --- cousin of axion--photon conversion (Primakoff).
      \end{itemize}
    \end{column}
