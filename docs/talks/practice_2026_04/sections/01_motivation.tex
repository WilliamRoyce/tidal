\begin{frame}{The Gertsenshtein channel for HFGW detection}
  
  \begin{itemize}\itemsep0.8em
    \item No GW detector at $\gtrsim$ MHz.  Hypothesised sources: early-universe processes, primordial-black-hole mergers and evaporation.

    \item \textbf{Gertsenshtein effect} (graviton $\leftrightarrow$ photon in a $B$-field).  Two routes explored: astrophysical-background reanalyses [1,\,2]; axion-haloscope reinterpretations [3,\,4].

    \item Bare GR mixing length $\sim 10^{6}$\,km even at magnetar fields.

    \item Structure beyond GR may amplify the conversion; not previously computed.
  \end{itemize}

  \vfill
  {\tiny [1] Lella \emph{et al.}, PRD \textbf{110} (2024).
  \quad [2] Domcke \& Garcia-Cely, PRL \textbf{126} (2021).
  \quad [3] Berlin \emph{et al.}, PRD \textbf{105} (2022).
  \quad [4] Domcke \emph{et al.}, JHEP \textbf{2022} (2022).}
\end{frame}
