\begin{frame}{\textsc{tidal}: the symbolic side}
  \note{\textasciitilde 10 min mark. Two key bits: component-level E-L (gives the speedup that makes scanning tractable), and the custom parallelisation framework called out in the abstract.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \textbf{Input:} a TOML file listing
      \begin{itemize}\scriptsize
        \item fields + Lagrangian terms,
        \item background fields,
        \item gauge fixing.
      \end{itemize}
      \vspace{0.2em}
      \textbf{Processing} (Mathematica + xAct / xTras):
      \begin{itemize}\scriptsize
        \item \texttt{xPert} perturbation around arbitrary backgrounds.
        \item Euler--Lagrange w.r.t.\ each field independently.
        \item Component decomposition, gauge fixing, coupling-matrix extraction.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \textbf{Two pieces of engineering that make this work:}
      \begin{itemize}\scriptsize
        \item \textbf{Component-level E--L} --- decompose to scalar components first, then differentiate. Skips abstract-tensor-index machinery. Derivations generating $\mathcal{O}(100)$ equations complete in seconds.
        \item \textbf{Custom parallelisation} --- xAct contraction kernel parallelised over independent symbolic sub-expressions. Without this, one Lagrangian variant takes a working day; with it, minutes.
      \end{itemize}
      \vspace{0.3em}
      \emph{Not a lookup table of hand-coded theories --- a symbolic compiler for arbitrary Lagrangians.}
    \end{column}
  \end{columns}
\end{frame}
