\begin{frame}{Torsion via Poincar\'e gauging (II)}
  \note{\textasciitilde 7 min mark. Geometric intuition + honest caveat about ghosts.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Geometric picture.} Parallel-transport around a closed loop has \emph{two} independent mismatches:
      \begin{itemize}
        \item \textbf{Rotational} mismatch $\to$ \emph{curvature}.
        \item \textbf{Translational} mismatch (loop fails to close) $\to$ \emph{torsion}.
      \end{itemize}
      \vspace{0.5em}
      \textbf{Why take it seriously?}
      \begin{itemize}\small
        \item Couples naturally to fermion spin (Einstein--Cartan).
        \item High-curvature / early-universe regimes $\Rightarrow$ PGT corrections potentially non-negligible.
        \item Minimal, well-motivated extension --- first place to look for BSM gravity.
      \end{itemize}
    \end{column}
