\documentclass[border=3pt,tikz]{standalone}
\usepackage{tikz}
\usepackage[svgnames]{xcolor}
\usetikzlibrary{decorations.markings,arrows.meta,calc}
\usepackage[outline]{contour} % glow around text
\contourlength{0.5pt}
\tikzset{>={Stealth[length=1.2mm,width=1.2mm]}} % default arrow tip for -> etc.
\colorlet{OrangeFill}{LightSalmon}
\colorlet{OrangeLine}{DarkSalmon!30!Chocolate}
\colorlet{GreenFill}{MediumAquamarine}
\colorlet{GreenLine}{MediumAquamarine!30!Teal}
\colorlet{PurpleFill}{MediumPurple}
\colorlet{PurpleLine}{MediumPurple!30!SlateBlue}
\colorlet{PinkFill}{Thistle}
\colorlet{PinkLine}{Thistle!30!Plum}
\colorlet{BlueFill}{LightSkyBlue}
\colorlet{BlueLine}{SteelBlue}
\colorlet{LimeFill}{YellowGreen}
\colorlet{LimeLine}{YellowGreen!30!Olive}
\colorlet{GrayFill}{LightGray!60!OldLace}
\colorlet{GrayLine}{RosyBrown!50!Silver}

\begin{document}

\newlength{\figL}
\setlength{\figL}{0.9cm}

\begin{tikzpicture}[
    x=\figL,
    y=\figL,
    line cap=round,
    line join=round,
    font=\normalsize
]

\pgfmathsetmacro{\Umax}{8.4}
\pgfmathsetmacro{\Vmax}{5.6}
\pgfmathtruncatemacro{\Nsamp}{20}

\pgfmathsetmacro{\sx}{0.046667}
\pgfmathsetmacro{\sy}{0.022500}

\pgfmathsetmacro{\axA}{0.036905}
\pgfmathsetmacro{\axB}{0.005000}
\pgfmathsetmacro{\ayA}{0.057143}
\pgfmathsetmacro{\ayB}{0.004821}

\pgfmathsetmacro{\kxA}{324.8}
\pgfmathsetmacro{\kxB}{649.6}
\pgfmathsetmacro{\kyA}{336.0}
\pgfmathsetmacro{\kyB}{672.0}
\pgfmathsetmacro{\phixA}{12}
\pgfmathsetmacro{\phixB}{-10}
\pgfmathsetmacro{\phiyA}{8}
\pgfmathsetmacro{\phiyB}{-15}

\def\X#1#2{(
    (#1)
    + \sx*\Umax*((#2)/\Vmax)
    + \axA*\Umax*sin(\kxA*((#2)/\Vmax) + \phixA)
    + \axB*\Umax*sin(\kxB*((#2)/\Vmax) + \phixB)
)}
\def\Y#1#2{(
    (#2)
    + \sy*\Vmax*((#1)/\Umax)
    + \ayA*\Vmax*sin(\kyA*((#1)/\Umax) + \phiyA)
    + \ayB*\Vmax*sin(\kyB*((#1)/\Umax) + \phiyB)
)}

\pgfmathsetmacro{\uxlabel}{0.18*\Umax}
\pgfmathsetmacro{\vxlabel}{0.82*\Vmax}

\pgfmathsetmacro{\uylabel}{0.78*\Umax}
\pgfmathsetmacro{\vylabel}{0.18*\Vmax}

\pgfmathsetmacro{\dblack}{0.4375*\Vmax}
\pgfmathsetmacro{\dblue}{0.31*\Umax}
\pgfmathsetmacro{\dred}{0.4018*\Vmax}
\pgfmathsetmacro{\ang}{0.04*min(\Umax,\Vmax)}

\fill[GrayFill, opacity=0.3]
  plot[variable=\t,domain=0:\Umax,samples=\Nsamp]
    ({\X{\t}{0}},{\Y{\t}{0}})
  --
  plot[variable=\t,domain=0:\Vmax,samples=\Nsamp]
    ({\X{\Umax}{\t}},{\Y{\Umax}{\t}})
  --
  plot[variable=\t,domain=\Umax:0,samples=\Nsamp]
    ({\X{\t}{\Vmax}},{\Y{\t}{\Vmax}})
  --
  plot[variable=\t,domain=\Vmax:0,samples=\Nsamp]
    ({\X{0}{\t}},{\Y{0}{\t}})
  -- cycle;

\foreach \k in {1,...,7}{
  \pgfmathsetmacro{\u}{\k*\Umax/8}
  \draw[PurpleLine, thin, loosely dashed]
    plot[variable=\t,domain=0:\Vmax,samples=\Nsamp]
      ({\X{\u}{\t}},{\Y{\u}{\t}});
}
\draw[GrayFill!30, thick, loosely dashed]
    plot[variable=\t,domain=0:\Vmax,samples=\Nsamp]
      ({\X{\Umax/2}{\t}},{\Y{\Umax/2}{\t}});

\foreach \k in {1,...,7}{
  \pgfmathsetmacro{\v}{\k*\Vmax/8}
  \draw[GreenLine, thin, loosely dashed]
    plot[variable=\t,domain=0:\Umax,samples=\Nsamp]
      ({\X{\t}{\v}},{\Y{\t}{\v}});
}
\draw[GrayFill!30, thick, loosely dashed]
    plot[variable=\t,domain=0:\Umax,samples=\Nsamp]
      ({\X{\t}{\Vmax/2}},{\Y{\t}{\Vmax/2}});

\draw[PurpleLine, thick, dashed]
    plot[variable=\t,domain=0:\Vmax,samples=\Nsamp]
      ({\X{\Umax/2}{\t}},{\Y{\Umax/2}{\t}});
\draw[GreenLine, thick, dashed]
    plot[variable=\t,domain=0:\Umax,samples=\Nsamp]
      ({\X{\t}{\Vmax/2}},{\Y{\t}{\Vmax/2}});

\coordinate (P) at ({\X{\Umax/2}{\Vmax/2}},{\Y{\Umax/2}{\Vmax/2}});

\draw[GrayLine, thick]
  plot[variable=\t,domain=0:\Umax,samples=\Nsamp]
    ({\X{\t}{0}},{\Y{\t}{0}})
  --
  plot[variable=\t,domain=0:\Vmax,samples=\Nsamp]
    ({\X{\Umax}{\t}},{\Y{\Umax}{\t}})
  --
  plot[variable=\t,domain=\Umax:0,samples=\Nsamp]
    ({\X{\t}{\Vmax}},{\Y{\t}{\Vmax}})
  --
  plot[variable=\t,domain=\Vmax:0,samples=\Nsamp]
    ({\X{0}{\t}},{\Y{0}{\t}})
  -- cycle;

\node[PurpleLine] at ({\X{\uxlabel}{\vxlabel}},{\Y{\uxlabel}{\vxlabel}})
  {\contour{GrayFill!30}{$x=\mathrm{const.}$}};

\node[GreenLine] at ({\X{\uylabel}{\vylabel}},{\Y{\uylabel}{\vylabel}})
  {\contour{GrayFill!30}{$y=\mathrm{const.}$}};

\draw[GrayFill!30, line width=1.5pt] (P) -- ++(102:{\dblack});
\draw[GrayFill!30, line width=1.5pt] (P) -- ++(-11:{\dblue});
\draw[GrayFill!30, line width=1.5pt] (P) -- ++(0,{-\dred});
\draw[GrayFill!30, line width=1.5pt] (P) -- ++({-\dred},0);

\draw[PurpleLine, thick,->]
  (P) -- ++(102:{\dblack})
  node[left] {\contour{GrayFill!30}{$\partial_{y}|_{P}$}};

\draw[GreenLine, thick,->]
  (P) -- ++(-11:{\dblue})
  node[right] {\contour{GrayFill!30}{$\partial_{x}|_{P}$}};

\draw[OrangeLine, thick,->]
  (P) -- ++(0,{-\dred})
  node[left] {$\mathbf e_{1}|_{P}$};

\draw[OrangeLine, thick,->]
  (P) -- ++({-\dred},0)
  node[left] {$\mathbf e_{2}|_{P}$};

\fill[GrayFill!30] (P) circle (2.5pt);
\fill[black!60!GrayLine] (P) circle (1.8pt) node[above right] {$P$};

\draw[OrangeLine, thick]
  ($(P)+(-\ang,0)$) -- ($(P)+(-\ang,-\ang)$) -- ($(P)+(0,-\ang)$);

\end{tikzpicture}

\end{document}
