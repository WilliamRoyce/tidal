  DashedArrows/.style 2 args={
    densely dashed, thin,
    postaction={decorate},

\begin{tikzpicture}[line cap=round]
    \def\Lx{1.44}
    \def\Ly{0.96}
    \begin{scope}[shift={(0,0)}]

        \def\Rx{0.7*\Lx}
        \def\Ry{0.45*\Ly}
        \coordinate (O) at (0,-0.2*\Ly);

        \fill[OrangeFill, opacity=0.4] (\Rx,0.3*\Ly) arc(0:180:{1.05*\Rx} and {0.95*\Ry}) -- (-1.1*\Rx,0.3*\Ly) to[out=-90,in=103] (-\Rx,-0.2*\Ly) arc(180:0:{\Rx} and {\Ry}) -- cycle;
        \fill[OrangeFill, opacity=0.4] ($(O)+(0,-\Ry)$) arc(-90:-180:{\Rx} and {\Ry}) to[out=103,in=-90] (-1.1*\Rx,0.3*\Ly) to[out=-90,in=180] ($(O)+(0,-\Ry)+(115:0.4*\Ly)$) -- cycle;

        \draw[OrangeLine, thick, ->] ($(O)+(0,-\Ry)$) --++ (115:0.4*\Ly);

        \node at (0,0) {$\mathcal{R}$};
    \end{scope}

    \begin{scope}[shift={(2*\Lx,0)}]
        \path[use as bounding box] (-\Lx,-\Ly) rectangle (\Lx,\Ly);
        \coordinate (O) at (-0.487*\Lx,-0.63*\Ly);
        \coordinate (R) at ($(O)+(98:0.9*\Lx)$);

        \draw[PurpleLine, thick, ->] (Q) -- (Qv);

        \fill[PurpleFill, opacity=0.2] (O) -- (R) to[bend left=15] (Qv) -- (Q) -- cycle;

        \draw[densely dashed, thin,->] (-0.45*\Lx,-0.95*\Ly) to[bend left=12] (-0.43*\Lx,0.99*\Ly);
        \draw[densely dashed, thin,->] (-0.8*\Lx,-0.7*\Ly) to[bend left=15] (0.8*\Lx,-0.9*\Ly);
