\newcommand{\Tmidlinelabel}[2]{
   \node[fill = white, inner sep = 0.2pt, outer sep=0.8pt, font=\footnotesize] (midlabel) at ($ (#1)!.5!(#2) $) {$\mathcal{T} \to 0$};
   \draw[decorate, decoration={zigzag,amplitude=1,segment length=4}, thick] (#1) --  (midlabel);
   \draw[decorate, decoration={zigzag,amplitude=1,segment length=4, post length=4}, thick, ->] (midlabel) -- (#2);
}
\newcommand{\Qmidlinelabel}[2]{
   \node[fill = white, inner sep = 0.2pt, outer sep=0.8pt, font=\footnotesize] (midlabel) at ($ (#1)!.5!(#2) $) {$\mathcal{Q} \to 0$};
   \draw[densely dashed, thick] (#1) --  (midlabel);
   \draw[densely dashed, thick, ->] (midlabel) -- (#2);
}
\newcommand{\Rmidlinelabel}[2]{
   \node[fill = white, inner sep = 0.2pt, outer sep=0.8pt, font=\footnotesize] (midlabel) at ($ (#1)!.5!(#2) $) {$\mathcal{R} \to 0$};
   \draw[decorate, decoration={snake,amplitude=1,segment length=10}, thick] (#1) --  (midlabel);
   \draw[decorate, decoration={snake,amplitude=1,segment length=10, post length=4}, thick, ->] (midlabel) -- (#2);
}
