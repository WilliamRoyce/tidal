% --- Custom macros ---

% Fields
\newcommand{\lag}{\mathscr{L}}           % Lagrangian density
\newcommand{\ham}{\mathscr{H}}           % Hamiltonian density
\newcommand{\field}[1]{\mathcal{#1}}     % field components: \field{H}_{xy}

% Operators
\newcommand{\lapl}{\nabla^2}             % Laplacian
\newcommand{\ddt}{\partial_t}            % time derivative
\newcommand{\ddx}{\partial_x}            % spatial derivatives
\newcommand{\ddy}{\partial_y}
\newcommand{\ddz}{\partial_z}

% Metrics
\newcommand{\gmunu}{g_{\mu\nu}}          % metric tensor
\newcommand{\etamunu}{\eta_{\mu\nu}}     % Minkowski metric
\newcommand{\sqrtg}{\sqrt{|g|}}          % metric determinant

% Perturbation theory
\newcommand{\pert}[1]{\delta #1}         % linear perturbation
\newcommand{\bg}[1]{\bar{#1}}            % background value

% Gertsenshtein
\newcommand{\Bzero}{B_0}                 % background magnetic field
\renewcommand{\kappa}{\varkappa}           % graviton-photon coupling
\newcommand{\Pconv}{P_\mathrm{conv}}     % conversion probability

% Torsion
\newcommand{\torsion}{T}                 % torsion tensor
\newcommand{\contorsion}{K}              % contorsion tensor
\newcommand{\ricci}{\tilde{R}}           % Ricci scalar with torsion
